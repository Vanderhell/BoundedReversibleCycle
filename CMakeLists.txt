cmake_minimum_required(VERSION 3.16)
project(BoundedReversibleCycle LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS NO)
add_library(brc STATIC src/brc.c)
target_include_directories(brc PUBLIC include)
function(strict_c target)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wsign-conversion -Wshadow -Wundef)
  endif()
endfunction()
strict_c(brc)
enable_testing()
foreach(name basic cycle vectors)
  add_executable(test_${name} tests/test_${name}.c)
  target_link_libraries(test_${name} PRIVATE brc)
  strict_c(test_${name})
  add_test(NAME ${name} COMMAND test_${name})
endforeach()
set_tests_properties(vectors PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/tests)
