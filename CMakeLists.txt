cmake_minimum_required(VERSION 3.16)
project(BoundedReversibleCycle LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS NO)
add_library(brc STATIC src/brc.c)
target_include_directories(brc PUBLIC include)
function(strict_c target)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wsign-conversion -Wshadow -Wundef)
  endif()
endfunction()
strict_c(brc)
enable_testing()
add_executable(test_basic tests/test_basic.c)
target_link_libraries(test_basic PRIVATE brc)
strict_c(test_basic)
add_test(NAME basic COMMAND test_basic)
